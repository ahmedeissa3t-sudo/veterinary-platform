function saveToLocal(item){ let arr=[]; try{ arr=JSON.parse(localStorage.getItem('my_vet_cases')||'[]'); if(!Array.isArray(arr)) arr=[]; }catch(e){ arr=[]; } arr.unshift({ id:item.id, doctor:item.doctor||'غير محدد', animal:item.animal||'غير محدد', owner:item.owner||'غير محدد', phone:item.phone||'', pet:item.pet||'بدون اسم', service:item.service||'غير محددة', date:item.date||'', time:item.time||'' }); localStorage.setItem('my_vet_cases',JSON.stringify(arr)); loadHistory(); } function loadHistory(){ const box=document.getElementById('historyList'); let arr=[]; try{ arr=JSON.parse(localStorage.getItem('my_vet_cases')||'[]'); if(!Array.isArray(arr)) arr=[]; }catch(e){ arr=[]; } if(!arr.length){ box.innerHTML='لا توجد طلبات مسجلة حتى الآن.'; return; } box.innerHTML=arr.map(x=>`
🔢 رقم الطلب: ${escapeHtml(x.id||'')}
👨‍⚕️ الطبيب: ${escapeHtml(x.doctor||'غير محدد')}
🐾 الحيوان: ${escapeHtml(x.animal||'غير محدد')}
🏥 الخدمة: ${escapeHtml(x.service||'غير محددة')} ${x.date?`
📅 الموعد: ${formatDate(x.date)}`:''} ${x.time?`
⏰ الساعة: ${formatTime(x.time)}`:''}
`).join(''); } function formatDate(date){ if(!date)return ''; const parts=String(date).split('-'); if(parts.length!==3)return date; return `${parts[2]}/${parts[1]}/${parts[0]}`; } function formatTime(time){ if(!time)return ''; const parts=String(time).split(':'); if(parts.length<2)return time; let h=parseInt(parts[0],10); const m=parts[1]; let period='صباحًا'; if(h>=12){ period='مساءً'; if(h>12)h-=12; } if(h===0)h=12; return `${String(h).padStart(2,'0')}:${m} ${period}`; } function escapeHtml(text){ return String(text??'') .replace(/&/g,'&') .replace(//g,'>') .replace(/"/g,'"') .replace(/'/g,'''); }
